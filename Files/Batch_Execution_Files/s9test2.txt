create table QuizMarks(rollNo STR, name STR, cgpa NUM, quizMarks NUM, location STR);
open table QuizMarks;
insert into QuizMarks values from s9quiz.csv;
SELECT * FROM QuizMarks INTO QuizMarksCG WHERE cgpa > 8;
open table QuizMarksCG;
SELECT * FROM QuizMarksCG INTO Scholars WHERE quizMarks > 85;
open table Scholars;
SELECT rollNo,name FROM Scholars INTO KeralaScholars WHERE location = Kerala;
SELECT rollNo,name FROM Scholars INTO GujaratScholars WHERE location = Gujarat;
SELECT rollNo,name FROM Scholars INTO TelanganaScholars WHERE location = Telangana;
